terraform - security group (r6i.4xlarge EC2)

provider "aws_provider" {
     region = "us-east-1"
}

 
resource "aws_security_group" "PathnexSG" {
  name        = "PathnexSG"
  description = "Pathnex SG for SSH"

  ingress {
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }


  resource "aws_instance" "pathnex-ex2" {
     ami                = "ami-0abcd1234abcd1234"
     instance_type      = "r6i.4xlarge"
     vpc_security_group_ids = [aws_security_group.PathnexSG.id]

  tags = {
    Name = "Pathnex-EC2"
  }
}

jenkins pipeline - nodejs build


pipeline {
  agent any

      stages {

        stage('chekout') {
           steps {
               git url: 'https://github.com/Pathnex/sample-node-app.git'
}
}

         stage('install dependencies') {
            steps {
                 sh 'npm install'
}
}
stage('Run Tests') {
            steps {
                sh 'npm test'
            }
        }
        stage('Build') {
            steps {
                sh 'npm run build'



}
}

}
}


GitLab CI — NodeJS Build

stages:
  - install
  - test
  - build

node-install:
  stage: install
  image: node:16
  script:
    - git clone https://github.com/Pathnex/sample-node-app.git
    - cd sample-node-app
    - npm install

node-test:
  stage: test
  image: node:16
  script:
    - cd sample-node-app
    - npm test

node-build:
  stage: build
  image: node:16
  script:
    - cd sample-node-app
    - npm run build
